# Plan 1: Foundation — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the repo, the Supabase schema with row-level security, the `record_sale` function, a Telegram bot skeleton, and a staff web app with login and menu editing, deployed to production.

**Architecture:** Postgres in Supabase is the ERP. Tables are readable only by staff through RLS, and writes that must snapshot prices go through `security definer` functions. Edge functions are thin Deno handlers around pure TypeScript modules in `supabase/functions/_shared/`, which Vitest tests in Node. The staff app is a static React + Vite build on Cloudflare Pages that talks to Supabase directly.

**Tech Stack:** Supabase (Postgres, Auth, Edge Functions, pgTAP via `supabase test db`), Supabase CLI via npm, Deno (edge runtime only), Node + npm workspaces, Vitest, React + Vite + TypeScript, React Router, zod, Testing Library, Cloudflare Pages, Telegram Bot API.

**Spec:** `docs/superpowers/specs/2026-09-30-operations-automation-design.md`. Roadmap: `docs/superpowers/plans/2026-09-30-roadmap.md`.

## Global Constraints

- Money is stored as integer **cents** (`*_cents`). Ingredient costs are integer **micros** of SGD per unit (`*_micros`, 1 SGD = 1,000,000) for sub-cent precision. The spec's `unit_cost_cents` is stored as `unit_cost_micros` for this reason.
- All times are `timestamptz`. Business logic uses the Asia/Singapore timezone.
- Every table in `public` has RLS enabled. Customers never get table access.
- Supabase project region: Singapore. Staff app hosting: Cloudflare Pages (free).
- Secrets live only in Supabase secrets, Cloudflare environment variables, or GitHub secrets. `.env*` files are git-ignored; commit `.env.example` only.
- No `console.log` in committed code. Errors shown to staff are plain-English sentences.
- Immutable data handling in TypeScript: return new objects, never mutate inputs.
- Files stay under 400 lines and functions under 50 lines.
- TDD: write the failing test, watch it fail, implement, watch it pass, commit.
- Conventional commits (`feat:`, `test:`, `chore:`, …). Do not push unless the owner asks.
- Before starting, read `PROGRESS.md` and check `git status`. Do not overwrite another assistant's uncommitted work.

```sql
create or replace function public.set_updated_at()
returns trigger language plpgsql set search_path = '' as $$
begin
  new.updated_at := now();
  return new;
end $$;

create table public.menu_items (
  id uuid primary key default gen_random_uuid(),
  name text not null check (length(trim(name)) between 1 and 60),
  category text not null default 'coffee' check (category in ('coffee', 'non_coffee', 'food')),
  price_cents integer not null check (price_cents >= 0),
  active boolean not null default true,
  on_preorder boolean not null default true,
  sold_out_today boolean not null default false,
  sort_rank integer not null default 0,
  tally_block_id uuid not null unique default gen_random_uuid(),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
create unique index menu_items_active_name on public.menu_items (lower(name)) where active;
create trigger menu_items_updated_at before update on public.menu_items
  for each row execute function public.set_updated_at();

create table public.modifiers (
  id uuid primary key default gen_random_uuid(),
  name text not null unique check (length(trim(name)) between 1 and 40),
  price_delta_cents integer not null default 0,
  active boolean not null default true
);

create table public.ingredients (
  id uuid primary key default gen_random_uuid(),
  name text not null unique,
  kind text not null check (kind in ('bean', 'milk', 'packaging', 'other')),
  unit text not null check (unit in ('g', 'ml', 'pc')),
  avg_cost_micros bigint not null default 0 check (avg_cost_micros >= 0),
  roaster text,
  bag_size_g integer check (bag_size_g > 0),
  lead_time_days integer check (lead_time_days >= 0),
  rest_days integer not null default 7 check (rest_days >= 0),
  waste_allowance_pct numeric(5,2) not null default 5 check (waste_allowance_pct between 0 and 100),
  active boolean not null default true,
  check (kind <> 'bean' or (bag_size_g is not null and lead_time_days is not null))
);

create table public.recipe_lines (
  menu_item_id uuid not null references public.menu_items on delete cascade,
  ingredient_id uuid not null references public.ingredients on delete restrict,
  qty numeric(10,2) not null check (qty > 0),
  primary key (menu_item_id, ingredient_id)
);

create table public.modifier_lines (
  modifier_id uuid not null references public.modifiers on delete cascade,
  ingredient_id uuid not null references public.ingredients on delete restrict,
```

```sql
create table public.staff_roles (
  user_id uuid primary key references auth.users on delete cascade,
  role text not null check (role in ('admin', 'staff'))
);
alter table public.staff_roles enable row level security;
create policy "read own role" on public.staff_roles
  for select to authenticated using (user_id = (select auth.uid()));

create or replace function public.is_staff()
returns boolean language sql stable security definer set search_path = '' as $$
  select exists (select 1 from public.staff_roles where user_id = (select auth.uid()));
$$;

create or replace function public.is_admin()
returns boolean language sql stable security definer set search_path = '' as $$
  select exists (select 1 from public.staff_roles
                 where user_id = (select auth.uid()) and role = 'admin');
$$;

revoke execute on function public.is_staff(), public.is_admin() from public, anon;
grant execute on function public.is_staff(), public.is_admin() to authenticated;

do $$
declare t text;
begin
  foreach t in array array['menu_items', 'modifiers', 'ingredients', 'recipe_lines', 'modifier_lines'] loop
    execute format('alter table public.%I enable row level security', t);
    execute format(
      'create policy "staff manage %1$s" on public.%1$I for all to authenticated
         using ((select public.is_staff())) with check ((select public.is_staff()))', t);
  end loop;
end $$;
```

```sql
returns table (ingredient_id uuid, qty numeric) language sql stable set search_path = '' as $$

returns public.orders language plpgsql security definer set search_path = '' as $$
declare
  v_order public.orders;
  v_item jsonb;
  v_menu public.menu_items;
  v_mods uuid[];
  v_qty integer;
  v_unit_price integer;

    raise exception 'not authorised' using errcode = '42501';
  end if;

    raise exception 'payment method must be paynow or cash' using errcode = '22023';

    raise exception 'a sale needs at least one item' using errcode = '22023';

      raise exception 'unknown or inactive menu item %', v_item ->> 'menu_item_id' using errcode = '22023';
    end if;
    if v_qty is null or v_qty < 1 or v_qty > 50 then
      raise exception 'quantity must be between 1 and 50' using errcode = '22023';
    end if;
    if (select count(*) from public.modifiers where id = any (v_mods) and active) <> cardinality(v_mods) then
      raise exception 'unknown, inactive or repeated modifier' using errcode = '22023';
    end if;

    v_unit_price := v_menu.price_cents
      + coalesce((select sum(price_delta_cents) from public.modifiers where id = any (v_mods)), 0);

    insert into public.order_items (order_id, menu_item_id, qty, modifier_ids, unit_price_cents, unit_cost_micros)

    from public.line_ingredient_usage(v_menu.id, v_mods) u;

    v_total := v_total + v_unit_price * v_qty;
  end loop;

  update public.orders set total_cents = v_total where id = v_order.id returning * into v_order;
  return v_order;
end $$;
```

```ts
export type TelegramUpdate = {
  update_id: number;
  message?: {
    chat: { id: number };
    from?: { first_name?: string };
    text?: string;
  };
};

export type Reply = { chatId: number; text: string };

export function verifySecret(header: string | null, expected: string): boolean {
  if (!header || !expected || header.length !== expected.length) return false;
  let diff = 0;
  for (let i = 0; i < expected.length; i += 1) {
    diff |= header.charCodeAt(i) ^ expected.charCodeAt(i);
  }
  return diff === 0;
}

export function replyFor(update: TelegramUpdate, allowedChatIds: readonly number[]): Reply | null {
  const msg = update.message;
  if (!msg?.text) return null;
  const chatId = msg.chat.id;
  const isStart = msg.text.trim() === '/start';

  if (!allowedChatIds.includes(chatId)) {
    return isStart
      ? { chatId, text: `This chat is not set up yet. Send this chat ID to the owner: ${chatId}` }
      : null;
  }
  if (isStart) {
    const name = msg.from?.first_name ?? 'there';
    return { chatId, text: `Hi ${name}, the café bot is connected. You will get orders and alerts here.` };
  }
  return null;
}
```

```ts
import { createClient } from 'npm:@supabase/supabase-js@2';
import { replyFor, verifySecret, type TelegramUpdate } from '../_shared/telegram.ts';

const BOT_TOKEN = Deno.env.get('TELEGRAM_BOT_TOKEN') ?? '';
const WEBHOOK_SECRET = Deno.env.get('TELEGRAM_WEBHOOK_SECRET') ?? '';
const supabase = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!);

async function allowedChatIds(): Promise<number[]> {
  const { data, error } = await supabase
    .from('settings').select('value').eq('key', 'telegram_allowed_chat_ids').single();
  if (error) throw new Error(`could not load Telegram allowlist: ${error.message}`);
  return Array.isArray(data.value) ? data.value.map(Number) : [];
}

async function send(chatId: number, text: string): Promise<void> {
  const res = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ chat_id: chatId, text }),
  });
  if (!res.ok) throw new Error(`Telegram sendMessage failed: ${res.status} ${await res.text()}`);
}

Deno.serve(async (req) => {
  if (!verifySecret(req.headers.get('x-telegram-bot-api-secret-token'), WEBHOOK_SECRET)) {
    return new Response('unauthorised', { status: 401 });
  }
  try {
    const update = (await req.json()) as TelegramUpdate;
    const reply = replyFor(update, await allowedChatIds());
    if (reply) await send(reply.chatId, reply.text);
    return new Response('ok');
  } catch (err) {
    console.error('telegram-webhook failed', err);
    return new Response('error', { status: 500 });
  }
});
```

```tsx
import { createContext, useContext, useEffect, useState, type ReactNode } from 'react';
import type { Session, SupabaseClient } from '@supabase/supabase-js';

export type StaffRole = 'admin' | 'staff';
export type AuthState =
  | { status: 'loading' }
  | { status: 'signedOut' }
  | { status: 'signedIn'; email: string; role: StaffRole | null };

export const AuthContext = createContext<AuthState>({ status: 'loading' });
export const useAuth = (): AuthState => useContext(AuthContext);

async function loadState(client: SupabaseClient, session: Session | null): Promise<AuthState> {
  if (!session) return { status: 'signedOut' };
  const { data, error } = await client
    .from('staff_roles').select('role').eq('user_id', session.user.id).maybeSingle();
  const role = !error && (data?.role === 'admin' || data?.role === 'staff') ? data.role : null;
  return { status: 'signedIn', email: session.user.email ?? '', role };
}

export function AuthProvider({ client, children }: { client: SupabaseClient; children: ReactNode }) {
  const [state, setState] = useState<AuthState>({ status: 'loading' });

  useEffect(() => {
    let cancelled = false;
    const apply = (session: Session | null) =>
      loadState(client, session).then((next) => { if (!cancelled) setState(next); });
    client.auth.getSession().then(({ data }) => apply(data.session));
    const { data: sub } = client.auth.onAuthStateChange((_event, session) => { void apply(session); });
    return () => { cancelled = true; sub.subscription.unsubscribe(); };
  }, [client]);

  return <AuthContext.Provider value={state}>{children}</AuthContext.Provider>;
}
```

```ts
import { z } from 'zod';

const PRICE_PATTERN = /^\d{1,3}(\.\d{1,2})?$/;

export function parsePriceToCents(input: string): number | null {
  const cleaned = input.trim().replace(/^S?\$/i, '');
  if (!PRICE_PATTERN.test(cleaned)) return null;
  const [dollars, cents = ''] = cleaned.split('.');
  return Number(dollars) * 100 + Number(cents.padEnd(2, '0'));
}

export function formatCents(cents: number): string {
  return `S$${(cents / 100).toFixed(2)}`;
}

export const CATEGORIES = ['coffee', 'non_coffee', 'food'] as const;

export const menuItemFormSchema = z.object({
  name: z.string().trim().min(1, 'Enter a name').max(60, 'Keep the name under 60 characters'),
  category: z.enum(CATEGORIES),
  price: z.string().transform((value, ctx) => {
    const cents = parsePriceToCents(value);
    if (cents === null) {
      ctx.addIssue({ code: 'custom', message: 'Enter a price like 5.50' });
      return z.NEVER;
    }
    return cents;
  }),
  onPreorder: z.boolean(),
});

export type MenuItemFormValues = z.input<typeof menuItemFormSchema>;
export type MenuItemInput = z.output<typeof menuItemFormSchema>;
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js';
import type { MenuItemInput } from './menuForm';

export type MenuItemRow = {
  id: string;
  name: string;
  category: 'coffee' | 'non_coffee' | 'food';
  price_cents: number;
  active: boolean;
  on_preorder: boolean;
  sold_out_today: boolean;
  sort_rank: number;
};

export type MenuApi = {
  list(): Promise<MenuItemRow[]>;
  create(input: MenuItemInput): Promise<void>;
  update(id: string, input: MenuItemInput): Promise<void>;
  setActive(id: string, active: boolean): Promise<void>;
};

const COLUMNS = 'id,name,category,price_cents,active,on_preorder,sold_out_today,sort_rank';

const toRow = (input: MenuItemInput) => ({
  name: input.name, category: input.category, price_cents: input.price, on_preorder: input.onPreorder,
});

function check(error: { message: string } | null, action: string): void {
  if (error) throw new Error(`Could not ${action}. ${error.message}`);
}

export function createMenuApi(client: SupabaseClient): MenuApi {
  return {
    async list() {
      const { data, error } = await client.from('menu_items').select(COLUMNS)
        .order('active', { ascending: false }).order('sort_rank').order('name');
      check(error, 'load the menu');
      return (data ?? []) as MenuItemRow[];
    },
    async create(input) {
      const { error } = await client.from('menu_items').insert(toRow(input));
      check(error, 'add the item');
    },
    async update(id, input) {
      const { error } = await client.from('menu_items').update(toRow(input)).eq('id', id);
      check(error, 'save the item');
    },
    async setActive(id, active) {
      const { error } = await client.from('menu_items').update({ active }).eq('id', id);
      check(error, active ? 'show the item' : 'hide the item');
    },
  };
}
```

```tsx
import { useState, type FormEvent } from 'react';
import { CATEGORIES, menuItemFormSchema, type MenuItemFormValues, type MenuItemInput } from './menuForm';

type Props = {
  initial: MenuItemFormValues;
  onSave: (input: MenuItemInput) => Promise<void>;
  onCancel: () => void;
};

export function MenuItemFormView({ initial, onSave, onCancel }: Props) {
  const [values, setValues] = useState<MenuItemFormValues>(initial);
  const [error, setError] = useState<string | null>(null);
  const set = (patch: Partial<MenuItemFormValues>) => setValues((v) => ({ ...v, ...patch }));

  async function onSubmit(e: FormEvent) {
    e.preventDefault();
    const parsed = menuItemFormSchema.safeParse(values);
    if (!parsed.success) {
      setError(parsed.error.issues[0]?.message ?? 'Check the form');
      return;
    }
    try {
      await onSave(parsed.data);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Could not save the item.');
    }
  }

  return (
    <form onSubmit={onSubmit} aria-label="Menu item">
      <label>Name<input value={values.name} onChange={(e) => set({ name: e.target.value })} /></label>
      <label>Category
        <select value={values.category} onChange={(e) => set({ category: e.target.value as MenuItemFormValues['category'] })}>
          {CATEGORIES.map((c) => <option key={c} value={c}>{c.replace('_', ' ')}</option>)}
        </select>
      </label>
      <label>Price (S$)<input inputMode="decimal" value={values.price} onChange={(e) => set({ price: e.target.value })} /></label>
      <label><input type="checkbox" checked={values.onPreorder} onChange={(e) => set({ onPreorder: e.target.checked })} />Show on pre-order form</label>
      {error && <p role="alert">{error}</p>}
      <button type="submit">Save</button>
      <button type="button" onClick={onCancel}>Cancel</button>
```

```tsx
import { useCallback, useEffect, useState } from 'react';
import type { MenuApi, MenuItemRow } from './menuApi';
import { formatCents, type MenuItemFormValues } from './menuForm';
import { MenuItemFormView } from './MenuItemFormView';

const EMPTY: MenuItemFormValues = { name: '', category: 'coffee', price: '', onPreorder: true };
const toForm = (row: MenuItemRow): MenuItemFormValues => ({
  name: row.name, category: row.category, price: (row.price_cents / 100).toFixed(2), onPreorder: row.on_preorder,
});

type Editing = { mode: 'new' } | { mode: 'edit'; row: MenuItemRow } | null;

export function MenuAdminPage({ api }: { api: MenuApi }) {
  const [items, setItems] = useState<MenuItemRow[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [editing, setEditing] = useState<Editing>(null);

  const reload = useCallback(() => {
    api.list().then(setItems, (err: Error) => setError(err.message));
  }, [api]);
  useEffect(reload, [reload]);

  if (editing) {
    return (
      <MenuItemFormView
        initial={editing.mode === 'edit' ? toForm(editing.row) : EMPTY}
        onCancel={() => setEditing(null)}
        onSave={async (input) => {
          await (editing.mode === 'edit' ? api.update(editing.row.id, input) : api.create(input));
          setEditing(null);
          reload();
        }}
      />
    );
  }

  return (
    <main>
      <h1>Menu</h1>
      {error && <p role="alert">{error}</p>}
      <button type="button" onClick={() => setEditing({ mode: 'new' })}>Add item</button>
      <table>
        <thead><tr><th>Item</th><th>Price</th><th>Pre-order</th><th /></tr></thead>
        <tbody>
          {items.map((row) => (
            <tr key={row.id} aria-disabled={!row.active}>
              <td>{row.name}</td>
              <td>{formatCents(row.price_cents)}</td>
              <td>{row.on_preorder ? 'Yes' : 'No'}</td>
              <td>
                <button type="button" onClick={() => setEditing({ mode: 'edit', row })}>Edit</button>
                <button type="button" onClick={() => api.setActive(row.id, !row.active).then(reload, (e: Error) => setError(e.message))}>
                  {row.active ? 'Hide' : 'Show'}
                </button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </main>
  );
}
```
